🏗️ 1. 项目架构概览

Physics-Supernova 是一个基于 AI Agent 的物理奥林匹克问题求解系统。它使用 smolagents 框架，结合多个专业工具来解决复杂的物理问题。

核心组件

组件 文件 功能
主程序 run.py 程序入口，负责参数解析、Agent创建和任务执行
Markdown工具 utils/markdown_utils.py 处理Markdown文件，解析文本和图像
图像分析工具 utils/imgTools.py 使用视觉模型分析物理图表
审查工具 utils/reviewTools.py 专家评审解决方案的准确性
记忆总结工具 utils/summemoryTools.py 总结已完成的部分答案
WolframAlpha工具 utils/wolframTools.py 执行数学计算和符号运算

def _create_Physics_agent(Tools_list, markdown_content, model_id,
                           managed_agents_list, max_steps=80, **kwargs):

    # 1. 初始化LLM模型
    model = LiteLLMModel(
        model_id=model_id,  # 默认: "openrouter/google/gemini-2.5-pro"
        api_key=os.environ.get("OPENROUTER_API_KEY"),
        api_base=os.environ.get("OPENROUTER_API_BASE"),
        max_completion_tokens=32768,
        num_retries=3,
        timeout=1200,
    )

    # 2. 实例化工具
    tools = []
    for tool in Tools_list:
        if tool.__name__ == "AskImageTool":
            tools.append(tool(vision_model_id=kwargs.get("image_tool_model")))
        elif tool.__name__ == "ReviewRequestTool":
            tools.append(tool(review_tool_model=kwargs.get("review_tool_model")))
        elif tool.__name__ == "SummarizeMemoryTool":
            tools.append(tool(summarize_model_id=kwargs.get("summarize_tool_model")))
        else:
            tools.append(tool())

    # 3. 配置Agent参数
    manager_agent_kwargs = dict(
        model=model,
        tools=tools,
        max_steps=max_steps,
        verbosity_level=2,
        name="physics_agent",
        managed_agents=managed_agents_list
    )

    # 4. 根据类型创建Agent
    if kwargs["manager_type"] == "CodeAgent":
        manager_agent_kwargs["additional_authorized_imports"] = [
            "os", "sys", "time", "matplotlib.pyplot", "numpy", "pandas"
        ]
        managerAgent = CodeAgent(**manager_agent_kwargs)
    else:
        managerAgent = ToolCallingAgent(**manager_agent_kwargs)

    # 5. 注册工具（建立双向引用）
    for toolName in managerAgent.tools:
        managerAgent.tools[toolName].worker_agent = managerAgent

    # 6. 设置高分辨率图像
    managerAgent.markdown_content_high_res_image = markdown_content

    return managerAgent

def forward(self, image_ref: str, question: str) -> str:
    """处理图像分析请求"""

    # 1. 提取图像字节数据
    img_bytes = self._extract_image_bytes(image_ref)

    # 2. 调整图像大小（如果需要）
    if self.max_short_side_pixels is not None:
        from PIL import Image
        from io import BytesIO

        img = Image.open(BytesIO(img_bytes))
        width, height = img.size
        if min(width, height) > self.max_short_side_pixels:
            scale = self.max_short_side_pixels / min(width, height)
            new_size = (int(width * scale), int(height * scale))
            img = img.resize(new_size, Image.Resampling.BILINEAR)

    # 3. 准备消息给视觉模型
    messages = [
        ChatMessage(role=MessageRole.SYSTEM,
                   content="You are an expert in dealing with image in Physics Olympiads."),
        ChatMessage(role=MessageRole.USER, content=[
            {"type": "image", "image": pil_img},
            {"type": "text", "text": question},
        ]),
    ]

    # 4. 调用视觉模型并返回结果
    resp = self.vision_model.generate(messages)
    return resp.content.strip()

def get_agent_memory_observation(agent: MultiStepAgent) -> str:
    """提取Agent的完整问题解决历史"""

    # 获取完整消息历史
    newest_input_messages = agent.write_memory_to_messages().copy()

def forward(self) -> str:
    """执行内存总结并重置"""

    # 1. 生成总结
    summary = summarize_agent_memory(self.summarize_model, agent=self.worker_agent)

    # 2. 创建新的内存步骤（基于初始任务）
    new_memory_step = deepcopy(memory.steps[0])
    if self.is_first_use:
        new_memory_step.task += (
            "You have finished a portion of the problem... "
            "Now continue to solve the rest parts..."
        )

    # 3. 清空内存并设置新起点
    self.worker_agent.memory.reset()
    self.worker_agent.memory.steps = []
    self.worker_agent.memory.steps.append(new_memory_step)

    # 4. 返回格式化的总结
    return f"Till now, you have finished some portion... Summary:\n{summary}"
⚠️ 重要：此工具会清空 Agent 的工作内存，因此只应在完成一个完整部分后调用。过早调用会导致上下文丢失。

def obtain_newContent_and_images_from_markdown(content: str, image_base_dir: str):
    """解析Markdown内容并提取图像"""

    IMG_FILE_EXTENSIONS = (".png", ".jpg", ".jpeg", ".gif", ".webp")
    image_pattern = re.compile(r'!\[([^\]]*)\]\(([^)]+)\)')
    images = {}
    new_content = content

    image_counter = 0
    for match in image_pattern.finditer(content):
        image_path = match.group(2)  # 获取路径部分
        if any(image_path.lower().endswith(ext) for ext in IMG_FILE_EXTENSIONS):
            image_counter += 1

            # 解析相对路径
            if not Path(image_path).is_absolute():
                full_image_path = Path(image_base_dir) / image_path
            else:
                full_image_path = Path(image_path)

            if full_image_path.exists():
                # 创建占位符 , , etc.
                current_key = f""
                images[current_key] = str(full_image_path)
                new_content = new_content.replace(match.group(0), current_key)

    return new_content, images

def obtain_task_and_images(input_markdown_file, tools_list,
                           managed_agents_list, manager_type):
    """获取任务字符串和图像"""

    # 1. 加载Markdown内容
    markdown_content = load_markdown_from_filepath(input_markdown_file)
    problem_text = markdown_to_plaintext(markdown_content)
    compressed_problem_images = markdown_images_compress(markdown_content, max_short_side_pixels=1080)

    # 2. 基础任务提示
    task = "Below is the full physics problem. If there are Images, Images are attached..."

    # 3. 根据工具列表添加提示
    if AskImageTool in ToolsList:
        task += "When you need to perform measurements on images, you MUST call the `ask_image_question` tool..."

    if ReviewRequestTool in ToolsList:
        task += "When you need expert review of your work, you MUST call the `ask_review_expert` tool..."

    if SummarizeMemoryTool in ToolsList:
        task += "When you are sure that you have finished a part of the problem, call `finalize_part_answer`..."

    # 4. 添加问题解决提示
    task += (
        "Your task is to solve the problem part by part, step by step. "
        "ONLY after you have FISHED the WHOLE PROBLEM should you call final_answer...\n\n"
        "PROBLEM STATEMENT (text with image placeholders):\n" + problem_text
    )

    return task, compressed_problem_images
💡 任务提示词的构成：
  1. 基础说明：告知Agent有图像附件和占位符
  2. 工具使用指南：根据可用工具动态添加使用说明
  3. 问题解决策略：逐步解决，完成后再调用final_answer
  4. 问题陈述：实际的物理问题文本（含图像占位符）

7.2 示例执行序列

针对示例问题的可能执行序列：

  1. Step 1: Agent 阅读问题，识别Part A需要WolframAlpha计算（但工具列表中没有）
  2. Step 2: 使用 CodeAgent 的代码执行能力计算积分
  3. Step 3: Part B 需要描述 Fig.1，调用 ask_image_expert(image_ref="<image_1>", question="请描述这个图像")
  4. Step 4: 获取图像专家的描述结果
  5. Step 5: 调用 ask_review_expert 审查Part A和Part B的答案
  6. Step 6: 根据审查反馈修正答案
  7. Step 7: 调用 finalize_part_answer() 总结Part A和Part B
  8. Step 8: 内存重置，继续Part C
  9. Step 9: Part C 需要描述 Fig.2，调用 ask_image_expert(image_ref="<image_2>", question="描述测量几何")
  10. Step 10: 完成所有部分后调用 final_answer

🎯 10. 关键要点总结

系统设计亮点

  • 模块化设计：工具、模型、Agent分离，易于扩展
  • 多模态支持：文本 + 图像处理能力
  • 动态工具加载：根据命令行参数灵活配置
  • 内存管理：支持长问题的分段处理和内存重置
  • 错误恢复：工具调用带重试机制

使用注意事项

  • 环境变量：需要设置 OPENROUTER_API_KEY
  • 图像格式：支持 .png, .jpg, .jpeg, .gif, .webp
  • 最大步数：默认80步，复杂问题可能需要调整
  • 内存消耗：高分辨率图像会增加内存使用
  • API成本：多模态模型调用成本较高

扩展建议

  • 添加新工具：在 utils/ 创建新工具类，在 TOOLNAME2TOOL 注册
  • 更换模型：通过 --manager-model 参数指定其他LLM
  • 自定义提示词：修改 obtain_task_and_images 中的提示构建逻辑
  • 日志记录：在工具的 forward 方法中添加日志
